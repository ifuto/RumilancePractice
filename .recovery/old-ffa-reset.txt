    public void setResetIntervalSeconds(int seconds) {
        applyResetInterval(Math.max(0, seconds), true);
    }

    private void applyResetInterval(int seconds, boolean persist) {
        this.resetIntervalSeconds = seconds;
        this.lastResetRemaining = Integer.MAX_VALUE;
        if (seconds <= 0) {
            this.nextResetAtMillis = 0L;
        } else {
            this.nextResetAtMillis = System.currentTimeMillis() + seconds * 1000L;
        }
        if (persist) {
            configService.ffa().set("settings.reset-interval-seconds", seconds);
            configService.save(ConfigService.FFA);
        }
    }

    private void tickReset() {
        if (resetIntervalSeconds <= 0 || nextResetAtMillis <= 0L) {
            return;
        }
        long now = System.currentTimeMillis();
        int remaining = (int) Math.max(0L, (nextResetAtMillis - now + 999L) / 1000L);
        if (remaining <= 0) {
            for (FfaArena arena : arenas.values()) {
                if (arena.enabled()) {
                    reset(arena.id());
                }
            }
            lastResetRemaining = Integer.MAX_VALUE;
            nextResetAtMillis = now + resetIntervalSeconds * 1000L;
            return;
        }
        int prev = lastResetRemaining;
        lastResetRemaining = remaining;
        for (int at : RESET_WARN_AT) {
            if (prev > at && remaining <= at) {
                announceReset(at);
            }
        }
    }

    private static final int[] RESET_WARN_AT = {300, 240, 180, 120, 60, 30, 5, 4, 3, 2, 1};

    private void announceReset(int remainingSeconds) {
        String time = remainingSeconds >= 60 && remainingSeconds % 60 == 0
                ? FfaResetTimes.format(remainingSeconds) + "."
                : remainingSeconds + (remainingSeconds == 1 ? " second." : " seconds.");
        for (FfaArena arena : arenas.values()) {
            if (!arena.enabled()) {
                continue;
            }
            Component message = Component.text("⚠ ", NamedTextColor.YELLOW)
                    .append(Component.text(arena.id() + " FFA will reset in ", NamedTextColor.WHITE))
                    .append(Component.text(time, NamedTextColor.YELLOW));
            Bukkit.broadcast(message);
        }
        for (Player player : Bukkit.getOnlinePlayers()) {
            soundService.play(player, "ffa-reset-warn");
        }
    }
